// app/api/subjects/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { getSubjectsForBoardGrade } from '@/lib/subjects-seed'

export async function GET(req: NextRequest) {
  const board = req.nextUrl.searchParams.get('board')
  const grade = req.nextUrl.searchParams.get('grade')

  if (!board || !grade) {
    return NextResponse.json({ error: 'board and grade are required' }, { status: 400 })
  }

  const subjects = getSubjectsForBoardGrade(board, grade)
  return NextResponse.json({ subjects })
}

## TASK 5 — API: GET /api/children/[childId]/subjects

Create this endpoint. Used by Assignments page to fetch the child's registered subjects:

```ts
// app/api/children/[childId]/subjects/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getServerSession } from 'next-auth' // or existing auth util
import { authOptions } from '@/lib/auth'

export async function GET(
  req: NextRequest,
  { params }: { params: { childId: string } }
) {
  const session = await getServerSession(authOptions)
  if (!session) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const childId = BigInt(params.childId)

  // Security: verify this child belongs to the logged-in parent
  const child = await prisma.child.findFirst({
    where: { id: childId, parentId: BigInt(session.user.parentId) }
  })
  if (!child) return NextResponse.json({ error: 'Not found' }, { status: 404 })

  const subjects = await prisma.childSubject.findMany({
    where: { childId },
    select: { subjectName: true },
    orderBy: { subjectName: 'asc' }
  })

  return NextResponse.json({
    subjects: subjects.map(s => s.subjectName),
    childName: child.name,
    grade: child.grade,
    board: child.board
  })
}
```
